def __init__(self, chor: Chor, name: str | None = None) -> None:
    """Prepare a recursion scope and its back-edge handle."""
    self._chor = chor
    if name is None:
        self._name = f'_loop_{chor._loop_serial}'
        chor._loop_serial += 1
    else:
        self._name = name
    self._scope = _Scope()
    self._handle = LoopHandle(chor, self._name)
